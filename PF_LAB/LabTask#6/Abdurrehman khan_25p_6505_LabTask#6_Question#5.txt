#include<stdio.h>
int main()
{
	char symbol;
	int total_rows;
	printf("Enter total rows:\n");
	scanf("%d", &total_rows);
	
	printf("Enter the symbol:\n");
	scanf("%c", &symbol);
	for(int i=1;i<=total_rows;i++)
	{
		for(int k=1;k<total_rows-i;k++)
		{
			printf(" ");
			
		}
		for(int j=1;j<2*i-1;j++)
	{
			printf("%c", symbol);
	}
	printf("\n");
}
return 0;
	}
