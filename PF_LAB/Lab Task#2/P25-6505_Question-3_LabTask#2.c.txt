#include<stdio.h>
int main()
{
    int salary;
    float tax;
    float tax_amount;
    float total_salary;

    printf("Enter your salary:\n");
    scanf("%d",&salary);
    printf("Enter your tax rate:\n");
    scanf("%f",&tax);
    tax_amount=salary*tax/100;
    total_salary=salary-tax_amount;
    printf("The total salary after deduction will be: %.2f\n", total_salary);
return 0;
}
