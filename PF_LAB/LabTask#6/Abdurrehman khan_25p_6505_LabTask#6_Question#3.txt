#include<stdio.h>
int main()
{
	int n,x;
	float power,fact,sum=0;
	printf("Enter the value of n:\n");
	scanf("%d", &n);
	printf("Enter the value of x:\n");
	scanf("%d", &x);
	for(int i=0;i<n;i++)
	{
		power=1;
		fact=1;
		for(int j=1;j<=i;j++)
		{
			power=power*x;
			fact=fact*j;
			
		}
		if(i==0)
		{
			sum=sum+1;
		}
		else{
				sum=sum+power/fact;
			
		}

	}
	printf("Sum=%.2f", sum);
return 0;
	
}
